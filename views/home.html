# 规则组合：重试 + 超时 + 回调
await (adapter.Send.To("user", "123")
       .Retry(3).Timeout(10)
       .Hook(on_success)
       .OnError(on_error)
       .Text("可靠发送"))

# Build 批量构建：统一执行
await (adapter.Send.To("group", "G001")
       .Build()
       .Text("第一句").Image("pic.jpg")
       .send_all())

统一作用域与生命周期

从模块、身份、出站三个维度统一控制可用范围；在框架运行的每个关键阶段插入自定义逻辑，模块卸载时注册的资源自动归属回收

归属权自动清理 生命周期钩子
config.toml 
[ErisPulse.scope.platforms.onebot11]
modules = ["Chat", "Tool*"]   # 白名单：精确/glob/正则
blocked = ["re:^Danger"]

[ErisPulse.scope.identity.users.onebot11]
allow = ["u_admin"]
deny = ["spam_*"]

[ErisPulse.scope.actions.MyModule]
send = { deny = true }        # 全禁发送
api  = { allow = ["get_*"] }   # 仅允许查询类 API
