from ErisPulse import sdk
# Рекомендуется корневой импорт (Python 3.10+): часто используемые символы импортируются напрямую из корневого пакета, но и глубокие пути по-прежнему поддерживаются
from ErisPulse import BaseModule, command

class Main(BaseModule):
    def __init__(self, sdk):
        self.sdk = sdk
        self.logger = sdk.logger.get_child("MyModule")
        self.storage = sdk.storage
    
    @staticmethod
    def get_load_strategy():
        """Возвращает стратегию загрузки модуля"""
        from ErisPulse.loaders import ModuleLoadStrategy
        return ModuleLoadStrategy(
            lazy_load=True,
            priority=0,
            depends=[],  # Необязательно: список модулей, от которых зависит данный модуль
            # Необязательно: ленивая активация по событиям — объявите триггеры, модуль автоматически загрузится при первом совпадении события/команды
            # activate_on=[{"command": {"name": "hello", "help": "Отправить приветствие"}}],
        )
    
    async def on_load(self, event):
        """Вызывается при загрузке модуля"""
        @command("hello", help="Отправить приветствие")
        async def hello_command(event):
            name = event.get_user_nickname() or "друг"
            await event.reply(f"Привет, {name}!")
        
        self.logger.info("Модуль загружен")
    
    async def on_unload(self, event):
        """Вызывается при выгрузке модуля"""
        self.logger.info("Модуль выгружен")

Метаинформация модуля

Note

Эта функция доступна начиная с ErisPulse 2.8.0+.

Метаинформация модуля — это описание того, для чего он предназначен, к какой категории относится и т.д. Метаинформация — это общие сведения о модуле, которые используются различными интерфейсами и экосистемными модулями, такими как модуль help, список модулей в Dashboard, магазин модулей и т.д.

Как и в случае с get_load_strategy(), возвращайте экземпляр класса ModuleMeta (с поддержкой типизации и автодополнения в IDE), но также допускается возвращать словарь:

class MyModule(BaseModule):
    @staticmethod
    def get_meta() -> ModuleMeta:
        return ModuleMeta(
            name="Погода",               # Отображаемое имя (по умолчанию имя регистрации)
            description="Получение погоды в городе",  # Описание модуля
            version="1.0.0",
            author="ErisDev",
            group="Инструменты",               # Категория функционала
            tags=["Погода", "Поиск"],
        )

Совместимый способ (через словарь):

class MyModule(BaseModule):
    @staticmethod
    def get_meta() -> dict:
        return {
            "name": "Погода",
            "description": "Получение погоды в городе",
            "version": "1.0.0",
            "author": "ErisDev",
            "group": "Инструменты",
            "tags": ["Погода", "Поиск"],
        }
  • module.get_meta("MyModule") читает уже разобранные метаданные (приоритет класса > регистрационной информации, автоматически дополняет имена команд модуля).
  • module.get_commands_overview() объединяет "метаинформацию модуля + зарегистрированные команды (псевдонимы/группы/описания)" и организует обзор всех команд по модулям.
  • Владелец команды можно получить через cmd_info["owner"] (автоматически вставляется контекстной системой при регистрации).

Поддержка i18n для полей метаинформации

Значения полей метаинформации могут быть простыми строками или словарями i18n {"i18n": "key.path", "default": "текст по умолчанию"} (согласно соглашению для description в конфигурации). Ключи перевода объявляются через I18nClass, а при чтении module.get_meta() автоматически разрешаются в текст текущего языка:

class MyModule(BaseModule):
    class I18nClass(BaseI18n):
        meta_description: I18nKey = I18nKey(
            default="Weather lookup",
            zh_CN="查询城市天气",
            en="Weather lookup",
        )

    @staticmethod
    def get_meta() -> ModuleMeta:
        return ModuleMeta(
            name="Погода",
            description={"i18n": "MyModule.meta_description", "default": "Weather lookup"},
        )

Обязательный к прочтению для L1 — одна из пяти основных документов в разделе "Путь чтения" на главной странице, необходима для разработки модуля.

В этой статье подробно объясняется полный жизненный цикл модуля от загрузки до выгрузки, методы объявления конфигурации и хранения данных — что делает за вас фреймворк, а что вам нужно написать самостоятельно. Прочитав статью, вы получите полное представление.

Жизненный цикл модуля

Стратегия загрузки

from ErisPulse.Core.Bases import BaseModule
from ErisPulse.loaders import ModuleLoadStrategy

class MyModule(BaseModule):
    @staticmethod
    def get_load_strategy():
        """Возвращает стратегию загрузки модуля"""
        return ModuleLoadStrategy(
            lazy_load=True,   # Загрузка по требованию или немедленная загрузка
            priority=0,       # Приоритет загрузки (чем больше, тем выше приоритет)
            depends=["OtherModule"]  # Необязательно: объявить зависимости от других модулей
        )

Модули, перечисленные в depends, если они не зарегистрированы, приведут к пропуску текущего модуля и записи предупреждения. Порядок загрузки определяется с помощью топологической сортировки, при равных приоритетах сортировка происходит по убыванию priority.

Note

Каскадное выгрузка / перезагрузка (ErisPulse 2.8.0+): при выгрузке модуля, от которого зависят другие модули, модули, зависящие от него, будут предварительно каскадно выгружены (в логах будет указан каскадный цепной путь); при горячей перезагрузке любого модуля (локальный плагин / модуль, зарегистрированный во время выполнения / пакет из PyPI), модули, от которых зависит этот модуль, также каскадно перезагружаются, чтобы избежать использования устаревших ссылок на экземпляры. Циклические зависимости будут отклонены с ошибкой RuntimeError при загрузке.

Перезагрузка — полная перезагрузка (ErisPulse 2.11.0+): reload() по умолчанию очищает кэш импорта по имени пакета и перезагружает зависимости, удаляя старые экземпляры до выгрузки (при ошибке отклоняется, старые экземпляры продолжают обслуживать), после успешной проверки подлинности кода (предупреждение о ложной перезагрузке / stale метка); при неудачной загрузке можно получить информацию о последней ошибке с помощью sdk.module.last_error(имя) (с traceback).

Метод on_unload

Вызывается при выгрузке модуля, используется для очистки ресурсов:

async def on_unload(self, event):
    # Очистка пользовательских ресурсов
    # sdk.client управляется фреймворком, закрывать вручную не нужно
    
    # Отмена обработчиков событий (фреймворк автоматически обрабатывает)
    self.logger.info("Модуль выгружен")

Детали создания и очистки фоновых задач (используя self.spawn() / фреймворк отменяет по умолчанию) см. в управлении жизненным циклом.

Метод on_before_unload

Note

Этот хук доступен начиная с ErisPulse 2.11.0+.

Опциональный хук, вызываемый на первом этапе цепочки выгрузки (до on_unload и освобождения ресурсов) — в этот момент все ресурсы модуля еще доступны, подходит для сохранения состояния в sdk.storage перед выгрузкой / сном / перезагрузкой, чтобы восстановить его при следующем on_load (продолжение сна при ленивой загрузке основано на этом):

async def on_before_unload(self, event):
    # event["reason"] указывает причину: unload / reload / disable / cascade / dormant
    if event["reason"] in ("reload", "dormant"):
        await self.sdk.storage.aset(f"{self._module_name}:state", self.collect_state())

Если хук завершается с ошибкой, это только предупреждение, не блокирует процесс выгрузки; модули без объявленного хука ведут себя как обычно. Семантика сна / пробуждения описана в Ленивой загрузке сна.

Выгрузка и полная выгрузка (purge)

Note

Эта функция доступна начиная с ErisPulse 2.8.0+.

unload() по умолчанию только отменяет загрузку (выгружает экземпляр и ресурсы), но сохраняет метаданные регистрации (класс модуля и метаинформация) — модуль всё ещё может быть обнаружен и перезагружен с помощью load(), без необходимости повторной register().

Для полной выгрузки (освобождение ссылки на класс модуля, очистка sys.modules, чтобы плагин и его уникальные зависимости могли быть собраны сборщиком мусора) передайте purge=True:

# Только отмена загрузки: сохраняются метаданные регистрации, можно перезагрузить в любое время
await sdk.module.unload("MyModule")

# Полная выгрузка: удаление метаданных регистрации + очистка sys.modules (только для плагинов из папки)
await sdk.module.unload("MyModule", purge=True)
Семантика unload() по умолчанию unload(purge=True)
Выгрузка экземпляра и ресурсов (события/задачи/маршруты/lifecycle/i18n) ✅ ✅
Сохранение метаданных регистрации (класс модуля и метаинформация) ✅ ❌ Удалено
Очистка sys.modules (только для плагинов из папки) ❌ ✅
Класс модуля может быть собран сборщиком мусора ❌ ✅
Перезагрузка load() доступен напрямую Необходимо сначала register() + load()

При purge=True зависимые модули, которые выгружаются каскадно, также будут удалены; после выгрузки фреймворк выполнит gc.collect() и проверит, можно ли собрать класс модуля/экземпляр, наличие оставшихся ссылок будет предупреждено в логах (с указанием источника, уровень DEBUG).

Что делает фреймворк при загрузке (вам нужно только написать on_load, остальное делается автоматически):

Этап Что делает фреймворк автоматически
Внедрение owner Во время инициализации модуля с помощью owner_scope — все зарегистрированные вами команды/события/хуки/фоновые задачи автоматически присваиваются к этому модулю, при выгрузке они удаляются по owner
Шаблон конфигурации Модули, объявившие ConfigClass, получают автоматически сгенерированный/заполненный конфигурационный раздел ErisPulse.<ModuleName>
Ключи перевода i18n Модули, объявившие I18nClass, автоматически регистрируют ключи перевода (автоматически удаляются при выгрузке)
Зависимости Сортировка по depends, гарантирует, что зависимые модули загружаются первыми; циклические зависимости отклоняются с ошибкой RuntimeError
SDK присвоение После инициализации присваивается к sdk.<ModuleName>, чтобы вы могли обращаться к sdk.MyModule.xxx

Что делает фреймворк при выгрузке (соответствует U1→U7): После выполнения on_unload происходит дополнительная очистка — фоновые задачи отменяются (созданные через self.spawn, для корректного завершения нужно сделать в on_unload), ключи перевода, маршруты, команды/обработчики событий, хуки lifecycle, затем удаляется свойство SDK. purge=True дополнительно удаляет метаданные регистрации и очищает sys.modules.

Именно эта автоматическая очистка — основа того, что «вам нужно только писать on_load/on_unload, не нужно вручную unregister» — фреймворк реализует "кто зарегистрировал, тот и очищает" как единый процесс.

Управление конфигурацией

Объявленная конфигурация (рекомендуется)

Начиная с версии v2.5.2, модуль может объявить класс конфигурации с помощью ConfigClass, используя ту же систему схемы конфигурации, что и адаптер. Конфигурация читается в реальном времени через self.cfg, изменения применяются немедленно:

from dataclasses import dataclass, field
from ErisPulse.Core.Bases import BaseModule
from ErisPulse.Core.Bases import BaseConfig

@dataclass
class MyModuleConfig(BaseConfig):
    api_key: str = field(
        default="",
        metadata={
            "description": {"i18n": "my_module.api_key", "default": "Ключ API"},
            "required": True,
            "secret": True,
            "ui": {"widget": "password", "group": "basic", "order": 1},
        },
    )
    timeout: int = field(
        default=30,
        metadata={
            "description": {"i18n": "my_module.timeout", "default": "Время ожидания (секунды)"},
            "ui": {"widget": "number", "group": "advanced", "order": 2},
        },
    )

class MyModule(BaseModule):
    ConfigClass = MyModuleConfig

    def __init__(self, sdk):
        self.sdk = sdk
        self.logger = sdk.logger.get_child("MyModule")

    async def on_load(self, event):
        self.logger.info("Модуль загружен")

    async def on_unload(self, event):
        pass

    async def do_something(self):
        cfg = self.cfg  # Чтение в реальном времени, тип безопасно
        api_key = cfg.api_key
        timeout = cfg.timeout

BaseConfig — универсальный базовый класс конфигурации, применим для адаптеров, модулей, внешних проектов и любых сценариев. Поля конфигурации поддерживают многоязычные описания i18n (см. документацию по i18n).

Note

Жизненный цикл модуля (включая адаптеры) — загрузка, выгрузка, запуск, остановка и т.д. — управляется фреймворком (модульный менеджер / scope предоставляет API для запроса и управления). ConfigClass должен содержать только бизнес-параметры, модули не должны вкладывать логику "нужно ли запускать себя" внутри.

Система схемы конфигурации также поддерживает (начиная с v2.8.0, см. основные понятия адаптера):

  • Автоматическое создание описания полей из docstring: если metadata description не объявлен, автоматически извлекается из docstring :ivar поле: описание или раздела Attributes:
  • Вложенные dataclass конфигурации: при типе поля вложенный dataclass, схема/шаблон/валидация обрабатываются рекурсивно, веб-интерфейс отображает вложенные группы
  • Поле example, не сохраняющееся на диск: поле с metadata={"example": True} не записывается в config.toml, только сохраняется в config.full.example (подходит для сложных, редко используемых настроек), пользователь может вручную установить и сохранить как обычное поле

Объявленные ключи перевода (v2.7.0+)

Начиная с v2.7.0, модуль может объявить ключи перевода, как и ConfigClass, с помощью вложенного класса I18nClass. Фреймворк автоматически регистрирует все объявленные ключи перевода при загрузке, без необходимости вызова i18n.register(), и регистрация происходит до генерации шаблона конфигурации, гарантируя, что ключи перевода, используемые в описании конфигурации, уже доступны.

from ErisPulse.Core.Bases import BaseConfig, BaseI18n, I18nKey

class MyModule(BaseModule):
    # Класс конфигурации (необязательно)
    @dataclass
    class ConfigClass(BaseConfig):
        welcome_msg: str = field(
            default="Добро пожаловать",
            metadata={
                "description": {"i18n": "mymodule.welcome_msg", "default": "Сообщение приветствия"},
            },
        )

    # Класс набора ключей перевода (необязательно)
    class I18nClass(BaseI18n):
        # Имя атрибута автоматически объединяется в полный ключ пути: <имя модуля>.<имя атрибута>
        welcome_msg: I18nKey = I18nKey(
            default="Welcome Message",   # Безопасный запасной текст без привязки к языку
            zh_CN="欢迎消息",
            zh_TW="歡迎訊息",
            en="Welcome Message",
            ja="ウェルカムメッセージ",
            ru="Приветственное сообщение",
        )
        hello: I18nKey = I18nKey(
            default="Hello, {name}!",
            zh_CN="你好，{name}！",
            zh_TW="你好，{name}！",
            en="Hello, {name}!",
            ja="こんにちは、{name}！",
            ru="Привет, {name}!",
        )

Подробнее см. Рекомендуемый способ написания i18n.

Ручное чтение конфигурации (устарело)

Устарело: используйте объявленную конфигурацию + чтение self.cfg в реальном времени.

class MyModule(BaseModule):
    def __init__(self, sdk):
        self.sdk = sdk

    def _load_config(self):
        config = self.sdk.config.getConfig("MyModule")
        if not config:
            self.sdk.config.setConfig("MyModule", {"api_key": "", "timeout": 30})
            return {"api_key": "", "timeout": 30}
        return config

async def handle_event(self, event):

Параметр event обработчика событий — это класс-обёртка Event (подкласс dict). Настоятельно рекомендуется добавить для него аннотацию типа:

from ErisPulse.Core.Event import Event

@message.on_private_message()
async def handler(event: Event):
    text = event.get_text()   # IDE автоматически дополнит все удобные методы
    await event.reply(text)   # Опечатки будут обнаружены уже при статической проверке

Без аннотации IDE не сможет распознать методы Event (подсказки для get_text() / reply() / wait_reply() / методов расширений платформы не появятся), и их придётся писать по памяти.

Обратите внимание на различие: event в колбэке обработчика событий — это класс-обёртка Event (аннотируется как Event); а event в методах жизненного цикла модуля on_load / on_unload — это обычный dict (аннотируется как dict). Не путайте их.

Ключевые особенности

  • Полная совместимость со словарём: Event наследуется от dict
  • Удобные методы: предоставляет множество удобных методов
  • Доступ через точку: поддерживает доступ к полям события через точку
  • Обратная совместимость: все методы являются необязательными

Функции ответа

Базовый ответ

  • reply(content, method="Text", at_sender=False, quote=False, at_users=None, reply_to=None, at_all=False, via=None, **kwargs) - универсальный метод ответа

    • content: отправляемое содержимое (текст, URL и т.д.)
    • method: метод отправки, по умолчанию "Text"; возможны "Image"/"Voice"/"Video"/"File" и т.д.
    • at_sender: упоминать ли отправителя (автоматически извлекает user_id)
    • quote: цитировать ли текущее сообщение в ответе (автоматически извлекает message_id)
    • at_users: список пользователей для @-упоминания, например ["user1", "user2"]
    • reply_to: ручное указание ID сообщения для ответа
    • at_all: упоминать ли всех участников
    • **kwargs: дополнительные параметры (например, user_id для метода Mention)

  • reply_ob12(message) - ответ с использованием сегментов сообщений OneBot12

    • message: список или словарь сегментов сообщений OneBot12, можно формировать с помощью MessageBuilder

Запрос возможностей платформы

  • supports(method) - проверяет, поддерживает ли текущая платформа определённый метод отправки (например, "Image", "Voice"), возвращает bool
  • available_methods() - перечисляет все доступные методы отправки текущей платформы, возвращает список имён методов

Функция пересылки

Примечание: функция пересылки должна быть реализована через Send DSL адаптера; сам класс-обёртка Event не предоставляет прямых методов пересылки.

Функции ожидания ответа

  • wait_reply(prompt=None, timeout=60.0, callback=None, validator=None, method="Text", pattern=None, regex=None) - ожидание ответа пользователя
    • prompt: подсказка-сообщение; если указана, будет отправлена пользователю
    • timeout: время ожидания до таймаута (в секундах), по умолчанию 60 секунд
    • callback: функция обратного вызова, выполняется при получении ответа
    • validator: функция проверки, используется для проверки валидности ответа
    • method: метод отправки подсказки, по умолчанию "Text"
    • pattern: glob-подстановочные знаки (* / ? / [seq]); текст ответа должен соответствовать, иначе ожидание продолжается
    • regex: регулярное выражение; текст ответа должен ему соответствовать (используется либо pattern, либо regex), иначе ожидание продолжается
    • возвращает объект Event с ответом пользователя, при таймауте возвращает None

Интерактивные методы

  • confirm(prompt=None, timeout=60.0, yes_words=None, no_words=None, method="Text", hint=False) - диалог подтверждения

    • возвращает True (подтверждение) / False (отрицание) / None (таймаут)
    • встроенное автоматическое распознавание слов подтверждения на китайском и английском языках, можно задать собственный набор слов
    • method: метод отправки, по умолчанию "Text"; поддерживается отправка подсказки нетекстовыми способами, такими как "Image"/"Markdown"
    • hint: добавлять ли в конец подсказки автоматически подсказку со словами подтверждения (например, "(да/нет)"), по умолчанию False

  • choose(prompt, options, timeout=60.0, method="Text", options_format="auto", merge_prompt=False, placeholder="{options}") - меню выбора

    • options: список текстов вариантов
    • возвращает индекс выбранного варианта (с нуля), при таймауте возвращает None
    • method: метод отправки, по умолчанию "Text"; текстовые методы (Text/Markdown/md/Html/h5) по умолчанию добавляют варианты в конец
    • options_format: формат вариантов (по умолчанию: "auto" — встроенный стиль выбирается автоматически в зависимости от method)
      • "auto": Markdown → маркированный список (- 1.вариант), Html → нумерованный список (<ol>), остальное → простой текстовый список
      • "list": по одному в строке, например 1. Вариант A\n2. Вариант B
      • "inline": в одну строку, например 1.A | 2.B
      • "md": маркированный список Markdown
      • "html": нумерованный список Html
      • callable: пользовательская функция, принимает list[str] и возвращает str
    • merge_prompt: принудительно объединять всё в одно сообщение при отправке, по умолчанию False
      • False (по умолчанию): текстовые методы объединяют автоматически; нетекстовые методы сначала отправляют prompt, затем варианты методом Text
      • True: при любом method всё объединяется в одно сообщение, которое отправляется указанным пользователем методом
    • placeholder: заполнитель для вставки вариантов, по умолчанию {options}; место в prompt, где встречается этот маркер, заменяется текстом вариантов; при задании пустой строки варианты всегда добавляются в конец

  • collect(fields, timeout_per_field=60.0) - сбор данных формы

    • fields: список полей; каждый элемент содержит key, prompt, необязательный validator и необязательный method
    • возвращает словарь {key: value}; при таймауте любого поля возвращает None
    • каждое поле поддерживает ключ method для указания метода отправки, например при сборе изображений: {"key": "avatar", "prompt": "Отправьте аватар", "method": "Image"}
    • каждое поле может содержать необязательный ключ options (список); при его наличии поле становится вопросом с выбором вариантов (автоматически вызывается логика choose)
    • каждое поле может содержать необязательные ключи options_format, merge_prompt, placeholder, управляющие форматом вариантов, поведением объединения сообщений и заполнителем

  • wait_for(event_type="message", condition=None, timeout=60.0) - ожидание произвольного события

    • condition: функция фильтрации; совпадение при возврате True
    • возвращает соответствующий объект Event, при таймауте возвращает None

  • conversation(timeout=60.0) - создание контекста многоходового диалога

    • возвращает объект Conversation, поддерживающий say()/wait()/confirm()/choose()/collect()/stop()
    • свойство is_active указывает, активен ли диалог

Методы расширений платформы

Адаптеры могут регистрировать специфичные для платформы методы у класса-обёртки Event. Методы доступны только на экземплярах Event соответствующей платформы; при обращении с других платформ выбрасывается AttributeError.

Методы платформы через Event.__getattribute__ имеют приоритет над встроенными методами, поэтому можно переопределять встроенные интерактивные методы, такие как confirm, choose, collect, wait_reply, предоставляя специфичные для платформы реализации (например, кнопки, карточки и т.д.). Встроенные реализации экспортируются в виде функций _builtin_* и доступны для вызова из переопределяющих методов.

# Событие электронной почты - только почтовые методы
event = Event({"platform": "email", "email_raw": {"subject": "Hello"}})
event.get_subject()      # ✅ возвращает "Hello"
event.get_chat_type()    # ❌ AttributeError

Кроссплатформенные расширения (подстановочный знак)

register_event_method и register_event_mixin поддерживают передачу "*" в качестве имени платформы; зарегистрированные таким образом методы доступны на экземплярах Event всех платформ. Подходит для функций, требующих кроссплатформенного переиспользования, таких как диалоги с ИИ, управление контекстом и т.д.

from ErisPulse.Core.Event.wrapper import register_event_method

@register_event_method("*")
async def ai_chat(self, prompt: str):
    # self — это экземпляр Event, с доступом к данным события и встроенным методам
    await self.reply(f"AI: {prompt}")

После регистрации обработчики событий любой платформы могут вызывать event.ai_chat(...).

Приоритет разрешения методов (от высшего к низшему): специфичные для платформы методы → методы с подстановочным знаком → встроенные методы → доступ по ключам словаря.

О том, как разработчики адаптеров регистрируют методы расширений, см. API системы событий — Кроссплатформенные расширения.

Tip

Note

Рекомендуется использовать self.spawn() (ErisPulse 2.8.0+). С версии 2.8.3 даже asyncio.create_task будет автоматически привязан к модулю (Task Factory автоматически регистрирует, отменяет при отключении, не утечка self); self.spawn() по-прежнему рекомендуется — поддерживает переключение потоков, явное указание owner=. До версии 2.8.3 фоновые задачи не привязаны, удерживают ссылку на self, что приводит к утечке при горячей перезагрузке; необходимо использовать self.spawn(). Подробнее смотрите в управление жизненным циклом.

Утилитарные модули: при управлении чужими объектами нужно обрабатывать "уведомления об отключении"

Когда это нужно: ваш модуль управляет объектами других модулей (периодические обратные вызовы, подписчики, соединения, кэш и т.д.). Если эти ссылки не будут удалены после отключения модуля, экземпляр модуля не может быть освобождён — это наиболее распространённая причина утечек памяти в утилитарных модулях.

from ErisPulse.Core.Bases import BaseModule
from ErisPulse.runtime import off_cleanup, on_cleanup

class MyToolModule(BaseModule):
    def __init__(self):
        self._entries = {}  # {имя модуля: управляемый объект}

    def register(self, entry):
        owner = on_cleanup(self._drop)   # ① При регистрации добавить в цепочку очистки, автоматически определить вызывающий модуль
        self._entries.setdefault(owner, []).append(entry)

    def _drop(self, owner: str):
        self._entries.pop(owner, None)   # ② При отключении модуля фреймворк автоматически вызывает: удалить его объекты

    async def on_unload(self, event):
        off_cleanup(self._drop)          # ③ Перед отключением модуля отменить хук

Вот и всё, фреймворк гарантирует:

  • При отключении / отключении модуля (или при закрытии адаптера) _drop("имя модуля") обязательно будет вызван
  • Определение вызывающей стороны полностью автоматическое: если вызывающий модуль в on_load вызывает sdk.MyToolModule.register(...), или через sdk.module.call("MyToolModule", "register", ...), будет правильно определён
  • Не нужно беспокоиться о времени — хук вызывается в цепочке очистки фреймворка, раньше диагностики утечек, не будет ложных срабатываний

Последствия отсутствия подключения: при полном отключении модуля экземпляр не может быть освобождён (диагностика утечек сообщает "неосвобождаемый"); если вызывающий модуль сам не отменяет уведомление в on_unload, утечка будет постоянной.

Обычным модулям (не управляющим чужими объектами) не нужно об этом беспокоиться — очистка ресурсов (команды / обработчики / маршрутизация / фоновые задачи и т.д.) автоматически управляется фреймворком.

Подробности о времени срабатывания, правилах определения вызывающей стороны, таймаутах и обработке ошибок смотрите в системе принадлежности · руководство для утилитарных модулей.

Обработка ошибок

1. Классификация обработки исключений

from ErisPulse.Core.Bases.errors import ClientError

async def handle_event(self, event: Event):
    try:
        result = await self._process(event)
    except ValueError as e:
        # Ожидаемые бизнес-ошибки
        self.logger.warning(f"Предупреждение: {e}")
        await event.reply(f"Ошибка параметра: {e}")
    except ClientError as e:
        # Ошибки сети (нижележащие aiohttp-исключения sdk.client уже преобразованы)
        self.logger.error(f"Ошибка сети {e.method} {e.url}: {e}")
        await event.reply("Сетевой запрос не удался, попробуйте позже")
    except Exception as e:
        # Неожиданные ошибки
        self.logger.error(f"Неизвестная ошибка: {e}", exc_info=True)
        await event.reply("Обработка не удалась, свяжитесь с администратором")
        raise

Слой моделей данных (ORM)

Tip

Когда стоит читать этот документ: если нужны модели данных, автоматическое создание таблиц или маппинг связей. Если нет — пропустите, это не повлияет на использование остальных функций.

Начиная с версии 2.9.0 во фреймворк встроен декларативный слой моделей данных: унаследуйте Model и объявите поля через Field, чтобы получить автоматическое создание таблиц и возможности CRUD. Модели построены непосредственно поверх встроенного слоя хранения — SQLite / MySQL / PostgreSQL определяется конфигурацией ErisPulse.storage.backend, при смене бэкенда не нужно изменять код моделей.

Объявление моделей

from ErisPulse.Core.Bases import Model, Field

class User(Model):
    id: int = Field(primary_key=True, autoincrement=True)
    name: str = Field(max_length=64)
    age: int = Field(default=0, ge=0, le=150)
    role: str = Field(default="user", choices=["user", "admin"])
    tags: list = Field(default_factory=list)          # JSON-столбец, автоматическая сериализация при чтении и записи
    bio: str = Field(default="", description={"i18n": "user.bio", "default": "Описание"})
  • Имя таблицы по умолчанию — имя класса в snake_case (UserProfile → user_profile); можно переопределить через __tablename__ = "xxx"
  • Набор ограничений Field совпадает с декларативным конфигурационным классом (choices / ge / le / max_length; description также поддерживает i18n-словарь); движок валидации имеет общую основу с валидацией конфигурации

Параметры полей

Параметр Описание
default Значение по умолчанию (не задано и не автоинкремент → обязательное NOT NULL)
default_factory Фабрика изменяемого значения по умолчанию (например, list)
primary_key Первичный ключ
autoincrement Автоинкрементный первичный ключ (подразумевает первичный ключ, автоматически заполняется после create)

column_type Переопределяет определение типа SQL-столбца

Создание таблиц и CRUD

await User.create_table()                      # Идемпотентно (IF NOT EXISTS)

user = await User.create(name="Alice", age=20) # Вставка (автоинкрементный первичный ключ заполняется автоматически)
got = await User.get(id=user.id)               # Первая запись по условию равенства

users = await User.where(User.age > 18).order_by("-age").limit(10).all()
first = await User.where(User.name == "Alice").first()
total = await User.count(User.age > 18)

user.age = 21
await user.save()                              # Обновление по первичному ключу (сначала проверка ограничений); при отсутствии первичного ключа выполняется вставка (автоинкрементный первичный ключ также заполняется в экземпляре)
await user.delete()                            # Удаление по первичному ключу

await User.update_all(User.age > 18, role="adult")  # Массовое обновление
await User.delete_all(User.age > 100)               # Массовое удаление

Выражения запросов поддерживают > >= < <= == !=, in_([...]), а также комбинации & (И) / | (ИЛИ):

await User.where((User.age > 18) & User.name.in_(["Alice", "Bob"])).all()

Транзакции

Чтение и запись ORM используют тот же маршрутизатор транзакций, что и слой хранения фреймворка: когда вы находитесь внутри контекстной транзакции storage.atransaction(), операции create / save / delete и запросы автоматически переиспользуют соединение транзакции; фиксация или откат выполняются единым образом вместе с транзакцией, отдельной фиксации не происходит.

async with storage.atransaction():
    await User.create(name="Alice")
    ...  # при исключении внутри блока INSERT выше также откатывается

Связь с декларативными конфигурационными классами

Объявления моделей разделяют одну и ту же основу с ConfigClass (@dataclass + field(metadata=...)) — набор ограничений, движок валидации (validate_field_constraints), реестр категорий типов (python_type_category); но базовые классы намеренно разделены: поля конфигурации — обычные значения (TOML round-trip, горячее обновление после однократной загрузки), поля моделей — дескрипторы столбцов (обращение к классу = выражение запроса, экземпляр на каждую строку). Один синтаксис объявления, две основы, каждая выполняющая свою роль.

Когда какой способ объявления использовать

Аспект Конфигурационный класс field(metadata=...) Поле модели Field()
Сценарий применения Параметры поведения модуля (немного, читаемо человеком, требуется горячее обновление) Записи бизнес-данных (много строк, программная запись/чтение, требуются запросы)
Форма хранения config.toml (с сохранением комментариев, TOML round-trip) Таблица базы данных (автоматическое создание таблиц, SQL-диалект)
Форма значения Обычное значение (прямой доступ к атрибуту dataclass) Дескриптор (обращение к классу = выражение столбца, обращение к экземпляру = значение строки)
Объявление ограничений metadata={"choices": ..., "min": ..., "max": ...} Field(choices=..., ge=..., le=..., max_length=...)
Общая основа Движок валидации + набор ограничений + реестр категорий типов (одни и те же) То же самое

Эмпирическое правило: «как работает модуль» — конфигурационный класс, «какие данные создал пользователь» — модель.

Ограничения и важные замечания

  • Бэкенд определяется глобальной конфигурацией хранения; модель может переопределить его через атрибут класса __storage__ с пользовательским экземпляром BaseStorage (для внедрения в тестах)
  • Поля list / dict (включая параметризованные дженерики, такие как list[int], dict[str, int]) хранятся как JSON-текстовые столбцы по категории контейнера, сериализация при чтении и записи выполняется автоматически
  • Перед записью (create / save) автоматически выполняется проверка ограничений; при неудаче выбрасывается ValueError (локализованное сообщение)
  • Автоматическая миграция реализована для сценария добавления столбцов (см. ниже); изменение типа столбца и удаление столбцов требуют ручной обработки
  • Поведение при сбое соединения с хранилищем совпадает с поведением слоя хранения: фреймворк не падает, после периода охлаждения выполняется автоматическое переподключение

Автоматическая миграция (этап 2)

create_table() при уже существующей таблице автоматически сравнивает существующие столбцы с полями модели: для новых полей автоматически выполняется ALTER TABLE ADD COLUMN (мигрируемые столбцы освобождаются от ограничения NOT NULL, существующие строки заполняются NULL), для новых полей с index=True синхронно создаётся индекс. Писать миграционные скрипты вручную не нужно.

# Эволюция модели после релиза v1: добавление полей email / bio
class User(Model):
    __tablename__ = "orm_users"

    id: int = Field(primary_key=True, autoincrement=True)
    name: str = Field(max_length=64)
    age: int = Field(default=0)
    email: str = Field(default="")     # Новое: при следующем create_table() автоматически выполнится ADD COLUMN
    bio: str = Field(default="")

await User.create_table()              # Идемпотентно: мигрируются только новые столбцы

Ограничения: поддерживается только добавление столбцов; изменение первичного ключа, изменение типа столбца и удаление столбцов требуют ручной обработки (во избежание случайных разрушительных ALTER-операций).

Внешние ключи (основа маппинга связей)

foreign_key="таблица.столбец" объявляет ограничение внешнего ключа на уровне столбца; в DDL генерируется клауза REFERENCES:

class Post(Model):
    __tablename__ = "posts"

    id: int = Field(primary_key=True, autoincrement=True)
    author: int = Field(foreign_key="orm_users.id")

    content: str = Field(max_length=255)

Маппинг связей (relationship)

Объявите relationship() как атрибут класса в теле класса модели; направление определяется автоматически в зависимости от того, в какой таблице объявлен столбец внешнего ключа:

from ErisPulse.Core.Bases import Model, Field, relationship

class User(Model):
    __tablename__ = "orm_users"

    id: int = Field(primary_key=True, autoincrement=True)
    name: str = Field(max_length=64)

    posts = relationship("Post", foreign_key="author")   # внешний ключ в таблице другой модели → has-many

class Post(Model):
    __tablename__ = "posts"

    id: int = Field(primary_key=True, autoincrement=True)
    author: int = Field(foreign_key="orm_users.id")
    content: str = Field(max_length=255)

    writer = relationship("User", foreign_key="author")  # внешний ключ в этой таблице → belongs-to

has-many: атрибут экземпляра возвращает набор запросов, доступны все цепочечные возможности QuerySet, create автоматически подставляет первичный ключ текущей таблицы в столбец внешнего ключа связанной модели:

alice = await User.get(name="Alice")

posts = await alice.posts.all()                          # все статьи этого пользователя
latest = await alice.posts.order_by("-id").first()
total = await alice.posts.count()
hot = await alice.posts.where(Post.content != "").all()  # добавление условия по полю связанной таблицы
await alice.posts.delete()                               # удаляются только записи этого пользователя

new_post = await alice.posts.create(content="hi")        # author автоматически = alice.id

belongs-to: прямое await возвращает экземпляр связанной модели (возвращает None, если совпадений нет или внешний ключ равен NULL):

post = await Post.get(id=1)
writer = await post.writer          # экземпляр User или None
await writer.posts.count()          # работает в обе стороны

Ключевые моменты:

  • В related передаётся строка с именем класса (разрешается лениво по реестру имён классов моделей, порядок определения сторон не важен) или непосредственно сам класс модели
  • Запросы по связям и модель другой стороны используют каждый свой бэкенд хранения (JOIN между разными бэкендами не поддерживается — запрос по связи выполняется как два независимых SQL-запроса)
  • Запросы по связям не кэшируются, каждый доступ — это немедленный запрос; при необходимости произвольных пользовательских запросов можно использовать User.where(...)

Что дальше

Нужно сменить базу данных или развернуться на облачной БД

Не нужна структура таблиц, нужно лишь хранить несколько пар ключ–значение

Conversation: многоходовой диалог

Tip

Когда читать этот документ: когда нужны конечный автомат многоходового диалога и восстановление из контрольных точек. Если это не нужно — пропустите, на использование остальных функций это не влияет.

Когда команда должна последовательно запрашивать у пользователя информацию в несколько шагов (сбор данных при регистрации, анкеты, мастер настройки), используйте Conversation, чтобы выстроить в одной сессии цепочку «вопрос → ожидание ответа → проверка → следующий шаг». После тайм-аута или перезапуска диалог можно восстановить из контрольной точки.

Создание диалога

Диалог создаётся методом conversation() объекта Event:

from ErisPulse.Core.Event import command

@command("quiz")
async def quiz_handler(event):
    conv = event.conversation(timeout=30)

    await conv.say("🎮 Добро пожаловать в викторину!")

    answer = await conv.choose("Вопрос 1: кто создатель Python?", [
        "Guido van Rossum",
        "James Gosling",
        "Dennis Ritchie",
    ])

    if answer is None:
        await conv.say("Время вышло, приходите в следующий раз!")
        return

    if answer == 0:
        await conv.say("Правильно!")
    else:
        await conv.say("Неправильно, правильный ответ — Guido van Rossum")

    conv.stop()

Основной API

say(content, **kwargs)

Отправляет сообщение, возвращает self, поддерживая цепочки вызовов:

await conv.say("Первая строка").say("Вторая строка").say("Третья строка")

Можно также указать метод отправки:

await conv.say("https://example.com/image.jpg", method="Image")

wait(prompt=None, timeout=None)

Ожидает ответ пользователя, возвращает объект Event или None (тайм-аут):

# Простое ожидание
resp = await conv.wait()
if resp:
    text = resp.get_text()

# Отправка подсказки и ожидание
resp = await conv.wait(prompt="Пожалуйста, введите ваше имя:")

# Своё время ожидания (переопределяет тайм-аут диалога по умолчанию)
resp = await conv.wait(prompt="Ответьте в течение 10 секунд:", timeout=10)

confirm(prompt=None, **kwargs)

Ожидает подтверждения пользователя (да/нет), возвращает True / False / None (тайм-аут):

result = await conv.confirm("Вы уверены, что хотите удалить все данные?")
if result is True:
    await conv.say("Удалено")
elif result is False:
    await conv.say("Отменено")
else:
    await conv.say("Время ожидания истекло, ответа нет")

Встроенные распознаваемые слова подтверждения: да/yes/y/подтверждаю/точно/хорошо/ok/true/верно/ага/ладно/согласен/без проблем/можно/конечно...

Встроенные распознаваемые слова отрицания: нет/no/n/отмена/не/не надо/нельзя/cancel/false/неверно/неправильно/не/отказ...

choose(prompt, options, **kwargs)

Ожидает, пока пользователь выберет один из вариантов, возвращает индекс варианта (начиная с 0) или None:

choice = await conv.choose("Выберите цвет:", ["Красный", "Зелёный", "Синий"])
if choice is not None:
    colors = ["Красный", "Зелёный", "Синий"]
    await conv.say(f"Вы выбрали {colors[choice]}")

Пользователь может выбрать, введя номер (1/2/3) или текст варианта (Красный).

options_format="auto" (по умолчанию) автоматически выбирает встроенный стиль в зависимости от method: Markdown→маркированный список, Html→нумерованный список, остальные→простой текстовый список. Также поддерживаются "list", "inline", "md", "html" или пользовательская функция.

Поддерживается merge_prompt=True для объединения подсказки и вариантов в одно сообщение, а также плейсхолдер, управляющий позицией вставки вариантов (по умолчанию {options}, настраивается через placeholder):

choice = await conv.choose(
    "## Выберите\n{options}",
    ["Вариант А", "Вариант Б"],
    method="Markdown",
    merge_prompt=True,
)

# Собственное имя плейсхолдера
choice = await conv.choose(
    "Выберите: [choices]",
    ["Вариант А", "Вариант Б"],
    placeholder="[choices]",
)

collect(fields, **kwargs)

Пошаговый сбор информации, возвращает словарь данных или None:

data = await conv.collect([
    {"key": "name", "prompt": "Введите имя"},
    {"key": "age", "prompt": "Введите возраст",
     "validator": lambda e: e.get("alt_message", "").strip().isdigit(),
     "retry_prompt": "Возраст должен быть числом, введите ещё раз"},
    {"key": "city", "prompt": "Введите город"},
])

if data:
    await conv.say(f"Регистрация успешна!\nИмя: {data['name']}\nВозраст: {data['age']}\nГород: {data['city']}")
else:
    await conv.say("Процесс регистрации прерван")

Конфигурация полей:

Параметр Описание По умолчанию
key Ключ поля (обязателен) -
prompt Подсказка "Введите {key}"
validator Функция проверки, принимает Event, возвращает bool нет
retry_prompt Подсказка при повторе после неудачной проверки "Некорректный ввод, попробуйте ещё раз"
max_retries Максимальное число повторных попыток 3
condition Функция условия, принимает dict уже собранных данных, возвращает bool нет

Условные поля: с помощью condition можно реализовать динамические формы — поле собирается только при выполнении условия:

data = await conv.collect([
    {"key": "has_car", "prompt": "У вас есть машина? (да/нет)"},

Управление активным состоянием

stateDiagram-v2
    state "Активен" as active
    state "Неактивен" as inactive
    [*] --> active: event.conversation()
    active --> active: say / wait / confirm / choose / collect
    active --> inactive: stop()
    active --> inactive: тайм-аут wait()
    active --> inactive: тайм-аут или исчерпание повторов в collect()

Диалог автоматически переходит в неактивное состояние в следующих случаях:

  1. Вызван метод stop()
  2. wait() по тайм-ауту вернул None
  3. collect() вернул None из-за тайм-аута на каком-либо шаге или исчерпания повторных попыток

После перехода в неактивное состояние все интерактивные методы (wait/confirm/choose/collect) немедленно возвращают None и больше не ожидают ввод пользователя.

Ветвления и переходы

Декоратор @conv.branch(name)

Метод branch() регистрирует ветвь диалога, а goto() выполняет переходы между ветвями:

@command("menu")
async def menu_handler(event):
    conv = event.conversation(timeout=60)

    @conv.branch("main")
    async def main_menu():
        await conv.say("=== Главное меню ===\n1. Личные данные\n2. Настройки\n3. Выход")
        resp = await conv.wait()
        if resp is None:
            return
        text = resp.get_text().strip()
        if text == "1":
            await conv.goto("profile")
        elif text == "2":
            await conv.goto("settings")
        elif text == "3":
            await conv.say("До свидания!")
            conv.stop()

    @conv.branch("profile")
    async def profile():
        await conv.say("=== Личные данные ===\nИмя: Alice\n0. Назад")
        resp = await conv.wait()
        if resp and resp.get_text().strip() == "0":
            await conv.goto("main")

    @conv.branch("settings")
    async def settings():
        await conv.say("=== Настройки ===\n1. Переключатель уведомлений\n0. Назад")
        resp = await conv.wait()
        if resp and resp.get_text().strip() == "0":
            await conv.goto("main")

    await conv.start()  # Начало с первой зарегистрированной ветви

conv.start(name=None)

Запускает диалог, по умолчанию — с первой зарегистрированной ветви:

await conv.start()           # С первой ветви
await conv.start("settings") # С указанной ветви

Контекст и персистентность

conv.context

У каждого экземпляра диалога есть встроенный словарь context для обмена состоянием между ветвями:

@conv.branch("step1")
async def step1():
    conv.context["username"] = resp.get_text().strip()
    await conv.goto("step2")

@conv.branch("step2")
async def step2():
    name = conv.context.get("username", "Неизвестно")
    await conv.say(f"Привет, {name}!")

save() / resume() / clear_saved()

Диалог поддерживает персистентность и может быть восстановлен после тайм-аута или прерывания:

# Сохранение состояния диалога (обычно вызывать вручную не нужно, см. «Автоматическое сохранение» ниже)
await conv.save()

# ... позже восстановление в той же сессии ...
conv2 = event.conversation()
if await conv2.resume():
    await conv2.say("С возвращением! Продолжим прерванный диалог")
else:
    await conv2.say("Предыдущий диалог не найден")

# Очистка сохранённого диалога
await conv.clear_saved()

Ключ хранения включает измерение target (conversation:{platform}:{user_id}:{target_id}), поэтому диалоги одного пользователя в разных сессиях не перезаписывают друг друга; архивы старого формата (без target) автоматически мигрируют при resume().

Автоматические контрольные точки и восстановление после перезапуска

Автоматическое сохранение

Фреймворк автоматически поддерживает контрольные точки в следующих случаях; обычно вручную вызывать save() не нужно:

Момент Поведение
Переход между ветвями goto() / start() Автосохранение (текущая ветвь + context)
stop() / тайм-аут wait() / сбой collect() Автоочистка (конечное состояние диалога)

TTL контрольных точек

Архивы содержат метку времени; архивы старше ErisPulse.interaction.checkpoint_ttl (по умолчанию 24 часа) удаляются:

  • Ленивое отбрасывание: если при восстановлении обнаруживается, что архив просрочен, он автоматически отбрасывается
  • Активная фоновая очистка: во фреймворке есть периодическая задача GC (лениво запускается после первого использования контрольных точек), которая активно перечисляет и удаляет просроченные архивы, чтобы при долгой работе просроченные контрольные точки не накапливались в storage бесконечно. Если после активной очистки архива в сессию придёт сообщение, оно обрабатывается как «контрольная точка отсутствует»

Автоматическое восстановление после перезапуска

После перезапуска фреймворка идущие диалоги (ожидающие корутины в памяти) теряются, но контрольные точки сохраняются. Если зарегистрировать фабрику восстановления через register_resume_handler, фреймворк после перезапуска при получении первого сообщения в этой сессии автоматически продолжит диалог:

from ErisPulse.Core.Event.wrapper import Conversation

@Conversation.register_resume_handler()  # можно передать platform="onebot11", чтобы ограничить платформу
def make_conversation(event) -> Conversation:
    # Задача фабрики: заново создать диалог и перерегистрировать все ветви
    conv = event.conversation(timeout=60)

    @conv.branch("menu")
    async def menu(conv, event):
        ...

    return conv

После регистрации, когда пользователь, находившийся до перезапуска в ветви menu, пришлёт первое сообщение, фреймворк автоматически: восстановит context → захватит это сообщение → продолжит диалог из сохранённой ветви. Без зарегистрированной фабрики этот механизм не создаёт накладных расходов.

Восстановление означает захват

При успешном resume() фреймворк автоматически делает две вещи:

  1. Захват сессии: автоматически выполняет acquire эксклюзивной аренды сессии — другие модули могут через sdk.interaction.get_owner_of(event) узнать, что «этот пользователь занят диалогом»; если сессия уже занята другим модулем, восстановление отменяется (возвращает False), чтобы два диалога не конфликтовали
  2. Возврат истории: последние 10 сообщений из входящего ящика сессии помещаются в conv.recent_history (контекст LLM у AI-модуля не обрывается после восстановления); можно отключить через resume(with_history=0)
if await conv.resume(with_history=20):
    for m in conv.recent_history:
        print(m["role"], ":", m["text"])

Ручное восстановление (если не использовать автоматический механизм)

@command("continue")
async def continue_handler(event):
    conv = event.conversation()
    # ... регистрация ветвей ...
    if await conv.resume():
        conv.goto(conv.get_current_branch())

Ожидание ответа: wait_reply

Одной фразой: приостанавливает текущую сопрограмму в ожидании следующего сообщения целевого пользователя; при совпадении возвращает событие ответа, при таймауте или отмене — None; pattern / regex / validator позволяют отсеивать неподходящие ответы.

wait_reply — краеугольный камень интерактивных сессий: он приостанавливает текущую сопрограмму и ждёт, пока целевой пользователь «ответит» следующим сообщением.

from ErisPulse.Core.Event import command

@command("ask")
async def ask_command(event):
    reply = await event.wait_reply(prompt="Введите ваше имя:", timeout=30)
    if reply is None:
        await event.reply("Время ожидания истекло")
        return
    await event.reply(f"Привет, {reply.get_text()}!")

Все параметры

Параметр Описание По умолчанию
prompt Подсказка, отправляемая перед приостановкой None
timeout Таймаут ожидания (секунды) 60
pattern glob-фильтр (* / ? / [seq]), при несовпадении ожидание продолжается None
regex Фильтр по регулярному выражению (при одновременном указании с pattern должны совпасть оба), при несовпадении ожидание продолжается None
validator Функция проверки (принимает Event, возвращает bool), при неудаче ожидание продолжается None
callback Колбэк при получении ответа (альтернативный вариант вместо возврата значения) None
method Метод отправки prompt "Text"

reply = await event.wait_reply(session=True, prompt="Кто-нибудь может подсказать?")

Граница между ожиданием и командами (сквозной проход команд)

Если во время ожидания поступает сообщение с зарегистрированной командой (например, /cancel), по умолчанию оно пропускается диспетчеру команд и не потребляется как ответ (параметр конфигурации ErisPulse.event.wait_reply.cmdpass; при True поведение инвертируется — все сообщения трактуются как ответы). Критерий разбора текста совпадает с диспетчеризацией команд: приоритет у alt_message, при его отсутствии — откат к конкатенации сегментов сообщения (с 2.10.1 действует и для адаптеров, использующих только message-сегменты).

Когда ожидание может быть отменено

Ожидание больше не сводится к «ожиданию таймаута» — в следующих случаях ожидание немедленно прекращается (wait_reply возвращает None), а не заставляет вызывающего ждать до таймаута:

Триггер Причина отмены (InteractionCancelled.reason) Описание
Модуль-владелец выгружен / отключён owner_unload Очистка по владению: чьё ожидание зарегистрировано, у того и изымается при исчезновении
Адаптер закрыт / перезапущен platform_stop Все приостановленные ожидания этой платформы отменяются
Сессия заменена новым ожиданием / арендой conflict См. «арбитраж сессий» ниже
Отвечающий в чёрном списке / модуль owner отвязан revoked Повторная проверка прав при совпадении ответа: измерение идентичности scope + измерение модуля
Совпадение с ответом пользователя —— Обычный путь, возвращает событие ответа

Базовое исключение — InteractionCancelled (входит в иерархию исключений InteractionError), wait_reply уже преобразует его в возврат None; вызывающим, которым нужна причина, стоит использовать низкоуровневый API sdk.interaction.register().

Полная цепочка определения совпадения ответа

При поступлении сообщения-ответа менеджер взаимодействия проверяет его в следующем порядке (выполняется до сопоставления команд, приоритет у непрерывности диалога — даже если сообщение уже занято другим обработчиком с высоким приоритетом, приостановленный диалог может завершиться):

Совпадение ключа сессии (точное измерение user → откат к уровню сессии)
  → Текстовая фильтрация pattern / regex (при несовпадении продолжаем ждать)
  → Проверка validator (при неудаче продолжаем ждать)
  → Повторная проверка прав (измерение идентичности scope + измерение модуля owner; при неудаче ожидание прекращается)
  → Пробуждение ожидающего + принятие события в обработку (mark_processed)

Декларативные многошаговые цепочки: .got() (2.11+)

@command возвращает агент потока управления для нескольких шагов; с помощью .got() пошагово декларируются этапы сбора — функции шагов по именам параметров получают уже собранные значения (включая входные параметры команды), типы автоматически определяются по type= / аннотациям. Низкоуровнево всё компилируется в Conversation: каждый шаг — отдельная ветвь, переходы автоматически сохраняют контрольные точки, и после перезапуска ответ пользователя продолжится с сохранённого шага (фабрика восстановления регистрируется автоматически при декорировании и автоматически снимается при выгрузке модуля).

@command("weather")
async def weather(event):
    pass  # Точка входа: prompt первого .got отправляется автоматически, вопрос задавать вручную не нужно

@weather.got("city", prompt="Ответьте названием города")
async def weather_city(event, city: str):
    await event.reply(f"{city}: ясно, 25℃")

@weather.got("days", prompt="Прогноз на сколько дней?", type=int, retry_prompt="Введите число")
async def weather_days(event, city: str, days: int):
    await event.reply(f"{city}, следующие {days} дней……")

Параметр Описание
name Имя собираемого значения (имя параметра, инжектируемого в функции последующих шагов; уникально в рамках потока, повторное объявление вызывает ошибку при регистрации)
prompt Вопрос, автоматически отправляемый при входе в шаг

Семантика и границы

  • Таймаут / прерывание на любом шаге → завершается весь поток, контрольные точки очищаются (семантика совпадает с wait_reply); внутри функций шагов можно в любой момент выводить промежуточные результаты через event.reply(...)
  • У одного пользователя одновременно может быть только одно приостановленное взаимодействие — ожидание второго потока вытеснит предыдущее (InteractionCancelled("conflict")); для взаимоисключающих подсказок используйте аренду сессии
  • Собранные значения существуют только внутри потока; для оркестрации со сложным состоянием, сохраняющимся между перезапусками, используйте напрямую Conversation (context + branch/goto + ручные контрольные точки)
  • @command возвращает объект-прокси: поведение вызова, __name__, docstring и пользовательские атрибуты полностью идентичны исходной функции; command.unregister(прокси) автоматически распаковывает объект

Таймеры сессии: remind / escalate

Превращает «таймаут» из возвращаемого значения в оркестрируемый примитив. Таймеры привязаны к интерактивной сессии, автоматически отменяются при выгрузке модуля / закрытии адаптера; лимит активных remind на одну сессию — 5.

remind: напоминание, если нет ответа

@command("ticket")
async def ticket_command(event):
    await event.reply("Тикет отправлен, результат обработки будет сообщён здесь")
    # Если 5 минут нет ответа — один раз мягко напомним; любой ответ пользователя автоматически его отменяет
    event.remind(300, "Вы ещё здесь? Как только будет результат — сразу сообщим")
    reply = await event.wait_reply(timeout=3600)
    ...
  • event.remind(delay, text=None, *, callback=None): по истечении срока отправляет text в текущую сессию (или вызывает callback(event); поддерживаются синхронные / асинхронные функции). Строгая проверка: необходимо указать что-то одно — text или callback (если не указано ни то ни другое, бросается ValueError)
  • Возвращает дескриптор Reminder: reminder.cancel() — ручная отмена, reminder.expired — проверка состояния
  • Автоматически отменяется после ответа пользователя в этой сессии — в этом суть семантики «напоминания»: напоминание появляется, только пока пользователь молчит
  • Доступно и внутри Conversation: conv.remind(120, "Всё ещё раздумываете?")

escalate: гарантированная эскалация по истечении срока

event.escalate(1800, lambda e: notify_master(f"Тикет не обрабатывался 30 минут: {event.get_command_args()}"))

Единственное отличие от remind: не отменяется ответом пользователя — эскалационное действие (уведомление владельца, передача живому оператору) является обещанием «гарантированной доставки по истечении таймаута», отменяется только вручную через cancel() / при выгрузке модуля / закрытии адаптера.

Многоканальное ожидание: expect + select

Одной фразой: expect лишь описывает ожидание, не регистрируя его; select разом приостанавливает несколько ожиданий и возвращает (индекс, событие ответа) по первому совпавшему, а несовпавшие автоматически очищаются.

Приостанавливает несколько ожиданий одновременно по принципу «кто первый — того и право» — типичные сценарии: одновременно ждать одобрения администратора и отзыва от пользователя, совместное голосование нескольких участников.

which, reply = await event.select(
    event.expect(pattern="Согласен*", user="10001"),
    event.expect(pattern="Отказ*", user="10002"),
    event.expect(validator=lambda e: e.get_text() == "Отложить", session=True),
    timeout=60,
)
if which is None:
    await event.reply("За 60 секунд не получено ни одного результата одобрения")
elif which == 0:
    await event.reply("Одобрено")
elif which == 1:
    await event.reply("Отклонено")
  • event.expect(...) создаёт описание ожидания (ничего не регистрируя): поддерживает pattern / regex / validator / user (ограничение круга отвечающих) / session (ответить может любой)
  • event.select(*expectations, timeout=60): регистрирует всё разом → при совпадении любого возвращает (индекс, событие ответа) → несовпавшие ожидания автоматически отменяются; при общем таймауте возвращает (None, None). Строгая проверка: должна быть передана хотя бы одна expectation, иначе бросается ValueError
  • Совпавшее событие уже принимается фреймворком в обработку (mark_processed) и не будет потреблено повторно другими обработчиками

{!--< tips >!--} По сравнению с ручной оркестрацией нескольких задач через asyncio.wait: select автоматически очищает несовпавшие ожидания, автоматически принимает совпавшие события в обработку, а повторная проверка прав и очистка по владению полностью работают — не нужно самостоятельно управлять никакими Future. {!--< /tips >!--}

Note

С версии 2.10.1 select — это агрегированное ожидание: все ожидания объединяются в одну запись + пользовательское правило совпадения, несколько путей с одним ключом (например, два разных pattern на одного user) больше не отменяют друг друга, возвращается первый совпавший путь. Критерий проверки текста ответа: приоритет alt_message + откат к сегментам сообщения (как при диспетчеризации команд).

Замечание о границах применимости: event.wait_for(...) подключается напрямую к шине событий адаптера и не проходит через промежуточное ПО и фильтрацию по условиям detail_type (допуск по идентичности страхуется на входе диспетчеризации); для ожиданий с полным конвейером используйте wait_reply / select.

Взаимоисключение сессий: acquire / hold / get_owner_of

Владение распространяется с «ресурсов» на «сессии» — «кем в данный момент занят этот пользователь» становится запросом первого класса.

# Запрос: кто сейчас взаимодействует в этой сессии? (при простое возвращает None)
owner = sdk.interaction.get_owner_of(event)
if owner and owner != "MyModule":
    return  # С другим модулем уже идёт диалог, не мешаем

# Эксклюзивная аренда: монопольный доступ к сессии (стратегия deny, при занятости возвращает None)
lease = sdk.interaction.acquire(event)          # TTL по умолчанию 1 час, можно передать ttl=
if lease is None:
    return  # Уже занята
try:
    ...  # Эксклюзивное взаимодействие
finally:
    lease.release()

Форма контекстного менеджера (при неудаче захвата бросается SessionOccupiedError):

with sdk.interaction.hold(event) as lease:
    ...  # Автоматическое освобождение при выходе

Аренда поддерживает продление через renew(ttl); TTL истекает лениво — просроченная аренда автоматически очищается при следующем обращении.

При восстановлении диалога через Conversation.resume() фреймворк автоматически захватывает аренду (см. «восстановление = перехват» в Conversation: многошаговые диалоги) — восстановленный диалог естественно владеет сессией, и другие модули не вмешаются.

Почтовый ящик сессии: event.history

Политику хранения (7 дней / лимит количества), API переопределения в рантайме, а также риски и аудит см. в специальной статье Почтовый ящик сессии.

Единая запись недавнего потока сообщений каждой сессии (обе стороны — пользователь + бот), служащая общим фундаментом фактов для модулей ИИ-контекста, защиты от повторов и анализа поведения — модулям больше не нужно хранить историю по отдельности.

messages = await event.history(20)   # Последние 20 сообщений текущей сессии, по возрастанию времени
for m in messages:
    print(m["role"], ":", m["text"])  # role: "user" / "bot"
  • Автоматическая запись: входящие сообщения (role=user) + исходящие тексты бота (role=bot)
  • Хранение: отдельная таблица SQLite, политика хранения = лимит на сессию (по умолчанию 50) + глобальный TTL (по умолчанию 7 дней)
  • Способ записи: буфер в памяти + фоновая пакетная запись на диск (задержка не более 1 секунды), интерфейсы запросов автоматически объединяют ещё не сброшенные буферные строки — «чтение своих записей» в рамках одного процесса не страдает; при штатном завершении (sdk.uninit / завершение процесса) выполняется автоматический сброс на диск. При жёстком сбое / принудительном завершении могут потеряться записи примерно за последнюю секунду: этот фундамент предназначен как кэш недавнего контекста и не подходит для персистентности аудиторского уровня
  • Конфигурация: ErisPulse.transcript = {enabled = true, max_per_session = 50, ttl_hours = 168}
  • API менеджера: sdk.transcript.append() / get() / clear(), aflush() / flush() — ручной сброс на диск

Транзакции сообщений: message_tx

Все исходящие отправки внутри транзакции автоматически заносятся в журнал; при аварийном выходе отправленные сообщения автоматически отзываются в обратном порядке (шаг пропускается, если адаптер не реализует delete_message, журнал при этом ведётся в обычном режиме).

async with event.message_tx():
    await event.reply("Обрабатываем, пожалуйста, подождите")
    result = await do_something()          # Здесь бросается исключение →
    await event.reply(f"Готово: {result}")   # Предыдущее «Обрабатываем» отзывается автоматически

Отправки вне транзакции не учитываются (нулевые накладные расходы); get_send_receipts() позволяет просмотреть чеки по отправкам текущей транзакции.

Сквозная трассировка: trace-id

Каждое входящее событие автоматически получает идентификатор трассировки (переиспользуется event["id"], при отсутствии генерируется), который сопровождает:

  • контекст handler-а (чтение через get_current_trace_id())
  • исходящие отправки (к строкам журнала [Send] добавляется [trace:...], поле trace_id в хуках message.sending/sent)
  • данные хуков жизненного цикла (в dict автоматически добавляется _trace_id)
  • направленные события (lifecycle.emit(..., to=...)) и чеки транзакций сообщений

Когда одно сообщение обрабатывается цепочкой из нескольких модулей, вся цепочка связывается одним и тем же ID (журналы / медленные запросы / аудит).

Связь с другими системами

  • Владение: ожидания / аренды / таймеры фиксируют owner, при выгрузке сразу изымаются (Система владения)
  • Область действия (scope): при совпадении ответа выполняется повторная проверка по измерениям идентичности + модуля; аудит межмодульных вызовов идёт по исходящему измерению (Область действия)
  • Conversation: многошаговый диалог — это машина состояний с ветвлениями над интерактивными сессиями (Conversation), его ожидания также подпадают под все семантики отмены / повторной проверки / владения, описанные на этой странице

Связанные документы

  • Conversation: многошаговые диалоги - машина состояний с ветвлениями, автоматические контрольные точки и восстановление после перезапуска
  • Система владения (owner) - полная картина очистки по владению и границы проектирования
